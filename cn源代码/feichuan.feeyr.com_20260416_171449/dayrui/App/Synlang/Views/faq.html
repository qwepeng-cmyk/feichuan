        function dr_cmf_tips(code, msg, time) {
            $.ajax({
                url: '/index.php?s=synlang&c=lang&m=tips&lang={SYN_LANG}',
                type: 'POST',
                data: { data: msg },
                async: false,
                success: function(json) {
                    if (json.data) {
                        dr_tips(code, dr_lang(json.data), time);
                        return;
                    }else{
                        dr_tips(code, dr_lang(msg), time);
                    }
                }
            });
